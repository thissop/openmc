\begin{frame}{The question}
  \begin{itemize}
    \item Spin-polarized fuel biases the DT neutron emission direction relative to the local field $B$.
    \item Idea: steer the fast-neutron load relative to the magnets.
    \item Two things could kill it --- a strongly-shaped 3D wall, and scattering in a thick blanket.
    \item Machinery validation on a public equilibrium.
  \end{itemize}
  \vfill
  \[
    I(\theta) \propto \sin^2\theta \ \ (\perp)
    \qquad\text{or}\qquad
    1 + 3\cos^2\theta \ \ (\parallel),
    \quad \theta \text{ from } B
  \]

\begin{frame}{The pipeline}
  \begin{itemize}
    \item Polarized CompiledSource --- sampler verified against the DT differential cross-section.
    \item Hand-built watertight DAGMC geometry to bypass a tooling blocker.
    \item OpenMC transport in two streams: free and scatter.

% ---------------------------------------------------------------- 11
\begin{frame}{Caveats and what's next}
  \begin{itemize}
    \item 10k-history smoke run; large MC error deep behind the blanket.
    \item Machinery validation, not a reactor claim.
    \item Naive constant-offset wall; small inboard self-fold at $\varphi = 0$ needs a proper offset.
    \item Next: variance reduction (weight windows / FW-CADIS) to converge $\eta$.
    \item Then an analytic wall-load cross-check, higher statistics, and physics equilibria.
  \end{itemize}
\end{frame}
